## Usage

```tsx
import { useVerifyDeviceCode } from "@better-auth-ui/react/plugins/device-authorization"

const verifyDeviceCode = useVerifyDeviceCode(authClient)

verifyDeviceCode.mutate({
  query: {
    user_code: "ABCD1234"
  }
})
```
