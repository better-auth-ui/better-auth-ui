<Step>
### Register the UI plugin

Pass `multiSessionPlugin()` to `<AuthProvider>`:

```tsx title="components/providers.tsx"
import { AuthProvider } from "@better-auth-ui/heroui"
import { multiSessionPlugin } from "@better-auth-ui/heroui/plugins/multi-session" // [!code highlight]

<AuthProvider
  authClient={authClient}
  navigate={navigate}
  plugins={[multiSessionPlugin()]} // [!code highlight]
>
  {children}
</AuthProvider>
```

</Step>
</Steps>
