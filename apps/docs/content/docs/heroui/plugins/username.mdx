### Register the UI plugin

Pass `usernamePlugin()` to `<AuthProvider>`:

```tsx title="components/providers.tsx"
import { usernamePlugin } from "@better-auth-ui/heroui/plugins/username" // [!code highlight]
import { AuthProvider } from "@better-auth-ui/heroui"

<AuthProvider
  authClient={authClient}
  navigate={navigate}
  plugins={[usernamePlugin()]} // [!code highlight]
>
  {children}
</AuthProvider>
```
