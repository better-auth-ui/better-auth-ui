<Step>
### Register the UI plugin

Pass `deleteUserPlugin()` to `<AuthProvider>`:

```tsx title="components/providers.tsx"
import { AuthProvider } from "@better-auth-ui/heroui"
import { deleteUserPlugin } from "@better-auth-ui/heroui/plugins/delete-user" // [!code highlight]

<AuthProvider
  authClient={authClient}
  navigate={navigate}
  plugins={[deleteUserPlugin()]} // [!code highlight]
>
  {children}
</AuthProvider>
```

</Step>
</Steps>
