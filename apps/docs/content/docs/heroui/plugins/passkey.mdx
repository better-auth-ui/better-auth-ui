### Register the UI plugin

Pass `passkeyPlugin()` to `<AuthProvider>`:

```tsx title="components/providers.tsx"
import { AuthProvider } from "@better-auth-ui/heroui"
import { passkeyPlugin } from "@better-auth-ui/heroui/plugins/passkey" // [!code highlight]

<AuthProvider
  authClient={authClient}
  navigate={navigate}
  plugins={[passkeyPlugin()]} // [!code highlight]
>
  {children}
</AuthProvider>
```

</Step>
