### Register the UI plugin

```tsx title="src/components/providers.tsx"
import { AuthProvider } from "@/components/auth/auth-provider"
import { dashPlugin } from "@/lib/auth/dash-plugin"
import { organizationPlugin } from "@/lib/auth/organization-plugin"

export function Providers(props: { children?: JSX.Element }) {
  return (
    <AuthProvider
      authClient={authClient}
      navigate={navigate}
      plugins={[organizationPlugin(), dashPlugin()]}
    >
      {props.children}
    </AuthProvider>
  )
}
```

## Runtime prerequisites

- Server: `dash()` from `@better-auth/infra`.
- Client: `dashClient()` from `@better-auth/infra/client`.
- Organization activity: the Better Auth organization server and client plugins.
- Runtime API: Dash hooks and query helpers from the [Solid package](/docs/solid).
