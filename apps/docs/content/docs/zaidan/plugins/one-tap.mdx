## Setup

<Steps>
<Step>

<include>../../../shared/ui/plugins/one-tap.mdx#shared-configure-the-better-auth-server-plugin</include>

```ts title="src/lib/auth.ts"
import { betterAuth } from "better-auth"
import { oneTap } from "better-auth/plugins" // [!code highlight]

export const auth = betterAuth({
  // ...
  plugins: [
    oneTap({
      clientId: process.env.GOOGLE_CLIENT_ID as string
    }) // [!code highlight]
  ]
})
```

</Step>
<Step>
### Configure the Better Auth client plugin

Use the same client ID in the Solid client:

```ts title="src/lib/auth-client.ts"
import { createAuthClient } from "@better-auth-ui/solid"
import { oneTapClient } from "better-auth/client/plugins" // [!code highlight]

export const authClient = createAuthClient({
  plugins: [
    oneTapClient({
      clientId: import.meta.env.VITE_GOOGLE_CLIENT_ID,
      promptOptions: {
        baseDelay: 1_000,
        maxAttempts: 3
      }
    }) // [!code highlight]
  ]
})
```

</Step>
<Step>
### Update the copied authentication views

Install the Solid registry item so the sign-in and sign-up surfaces include the headless prompt slot:

```bash
bunx --bun shadcn@latest add https://better-auth-ui.com/r/solid/one-tap.json
```

</Step>
<Step>
### Register the Solid UI plugin

<include>../../../shared/ui/plugins/one-tap.mdx#shared-register-the-better-auth-ui-plugin</include>

```tsx title="src/components/providers.tsx"
import { oneTapPlugin } from "@better-auth-ui/solid/plugins/one-tap" // [!code highlight]

<AuthProvider
  authClient={authClient}
  navigate={navigate}
  plugins={[
    oneTapPlugin() // [!code highlight]
  ]}
  socialProviders={["google"]}
>
  {children}
</AuthProvider>
```

</Step>
