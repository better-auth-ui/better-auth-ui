## Setup

<Steps>
<Step>
### Configure the Better Auth server plugin

Add Better Auth's [Anonymous](https://www.better-auth.com/docs/plugins/anonymous) plugin:

```ts title="src/lib/auth.ts"
import { betterAuth } from "better-auth"
import { anonymous } from "better-auth/plugins" // [!code highlight]

export const auth = betterAuth({
  // ...
  plugins: [
    anonymous() // [!code highlight]
  ]
})
```

<include>../../../shared/ui/plugins/anonymous.mdx#shared-migration</include>

</Step>
<Step>
### Configure the Better Auth client plugin

Add `anonymousClient()` to the Solid client:

```ts title="src/lib/auth-client.ts"
import { createAuthClient } from "better-auth/solid"
import { anonymousClient } from "better-auth/client/plugins" // [!code highlight]

export const authClient = createAuthClient({
  plugins: [
    anonymousClient() // [!code highlight]
  ]
})
```

</Step>
<Step>
### Install the UI plugin

Install the Solid button and local plugin factory:

```bash
bunx --bun shadcn@latest add https://better-auth-ui.com/r/solid/anonymous.json
```

<include>../../../shared/ui/plugins/anonymous.mdx#shared-install-the-ui-plugin</include>

</Step>
<Step>

<include>../../../shared/ui/plugins/anonymous.mdx#shared-register-the-plugin</include>

```tsx title="src/components/providers.tsx"
import { AuthProvider } from "@/components/auth/auth-provider"
import { anonymousPlugin } from "@/lib/auth/anonymous-plugin" // [!code highlight]

<AuthProvider
  authClient={authClient}
  navigate={navigate}
  plugins={[
    anonymousPlugin() // [!code highlight]
  ]}
>
  {props.children}
</AuthProvider>
```

</Step>
</Steps>
