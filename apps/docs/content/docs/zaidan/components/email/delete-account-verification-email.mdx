```tsx
sendDeleteAccountVerification: async ({ user, url }) => {
  const html = await render(() =>
    DeleteAccountVerificationEmail({
      url,
      email: user.email,
      appName: "My App"
    })
  )

  await sendEmail({
    to: user.email,
    subject: "Confirm account deletion",
    html
  })
}
```
