## Usage

```tsx
import { parseOAuthAuthorizationRequest } from "@better-auth-ui/core/plugins/oauth-provider"
import { usePublicOAuthClient } from "@better-auth-ui/solid/plugins/oauth-provider"

const client = usePublicOAuthClient(
  authClient,
  () => parseOAuthAuthorizationRequest(window.location.search).clientId
)
```

The query remains disabled until the client ID accessor returns a value.
