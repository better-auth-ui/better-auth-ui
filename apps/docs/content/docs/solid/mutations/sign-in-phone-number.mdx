## Options factory

```tsx
import {
  type PhoneNumberAuthClient,
  signInPhoneNumberOptions,
  useAuth
} from "@better-auth-ui/solid"
import { createMutation } from "@tanstack/solid-query"

const { authClient } = useAuth()
const signIn = createMutation(() =>
  signInPhoneNumberOptions(authClient as PhoneNumberAuthClient)
)

signIn.mutate({
  phoneNumber: "+12025550123",
  password,
  rememberMe: true
})
```

With Better Auth `requireVerification`, an unverified credential returns `PHONE_NUMBER_NOT_VERIFIED` and starts verification.
