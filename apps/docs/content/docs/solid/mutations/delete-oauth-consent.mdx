## Usage

```tsx
import { useDeleteOAuthConsent } from "@better-auth-ui/solid/plugins/oauth-provider"

const deleteConsent = useDeleteOAuthConsent(authClient)

for (const id of application.consentIds) {
  await deleteConsent.mutateAsync({ id })
}
```

import { deleteOAuthConsentOptions } from "@better-auth-ui/core/plugins/oauth-provider"
